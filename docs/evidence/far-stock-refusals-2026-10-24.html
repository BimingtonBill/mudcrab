WITH p AS (
 SELECT reference_id, COUNT(*) AS n, SUM(status='clustered') AS clustered
 FROM pkg.cluster_reference_coverage WHERE scope='primitive'
 GROUP BY reference_id
)
SELECT r.id, r.base_form_id, rec.record_type, s.model_path,
       r.pos_x, r.pos_y, cov.status, cov.reason,
       COALESCE(p.n,0), COALESCE(p.clustered,0), cov.model_hash
FROM "references" r
JOIN records rec ON rec.form_id=r.base_form_id
LEFT JOIN statics s ON s.id=r.base_form_id
LEFT JOIN pkg.cluster_reference_coverage cov
 ON cov.reference_id=r.id AND cov.scope='reference'
LEFT JOIN p ON p.reference_id=r.id
WHERE r.is_exterior=1 AND r.worldspace_id=60
 AND rec.record_type IN (SELECT value FROM json_each(?));

SELECT p.reference_id,p.model_hash,p.primitive_index,p.reason,j.detail
FROM pkg.cluster_reference_coverage p
LEFT JOIN pkg.cluster_rejects j
 ON j.mesh_hash=p.model_hash AND j.primitive_index=p.primitive_index
WHERE p.scope='primitive' AND p.status='rejected';
